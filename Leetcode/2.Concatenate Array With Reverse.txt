//leetcode 

//2.Concatenate Array With Reverse

//Input: nums = [1,2,3]
Output: [1,2,3,3,2,1]

Explanation:
The first n elements of ans are the same as nums.
For the next n = 3 elements, each element is taken from nums in reverse order:

//solution 
class Solution {
public:
    vector<int> concatWithReverse(vector<int>& nums) {
        int n=nums.size();
        vector<int> ans(2*n);
        for (int i=0; i<n;i++){
            ans[i] = nums[i];
            ans[i + n] = nums[n - i - 1];
        }
        return ans;
    }
};
