// leetcode 

//2 Reverse string

//input 
Input: s = ["h","e","l","l","o"]
Output: ["o","l","l","e","h"] 

//solution 
class Solution {
public:
    void reverseString(vector<char>& s) {
        int left=0; 
        int right=s.size()-1;
        while(left<right){
            swap(s[left],s[right]);
            left++;
            right--;    
        }
        
    }
};
