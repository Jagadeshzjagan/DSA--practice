//leetcode 

//6.Single Number 2 

//Input: nums = [2,2,3,2]
Output: 3

//soluion 

class Solution {
public:
    int singleNumber(vector<int>& nums) {
        unordered_map<int,int> freq;
        for (int i = 0; i < nums.size(); i++) {
            freq[nums[i]]++; 
        }
        for (auto pair:freq){
            if (pair.second==1){
                return pair.first;
            }
        }
        return -1;
    }   
};
