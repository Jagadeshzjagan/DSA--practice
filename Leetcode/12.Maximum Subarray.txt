//leetcode 

//12.Maximum Subarray 

//Example 1:

Input: nums = [-2,1,-3,4,-1,2,1,-5,4]
Output: 6
Explanation: The subarray [4,-1,2,1] has the largest sum 6.

Example 2:
Input: nums = [1]
Output: 1
Explanation: The subarray [1] has the largest sum 1. 

//solution 
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
         long long sum=0; 
         long long max=LONG_MIN;
         int ans_start=-1; 
         int ans_end=-1;
         int start;
         for (int i=0; i<nums.size() ; i++){
            if (sum==0) {
                start=nums[i];
            }
             sum+=nums[i];
            if (sum>max){
                max=sum;
                ans_start=start, ans_end=nums[i];
            }
            if(sum<0){
                sum=0;
            }
        }
     return max;
    }
    
};
